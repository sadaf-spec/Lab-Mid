#include <stdio.h>
#include <stdlib.h>
#include <string.h>


struct Student {
   char name[50];
   int marks;
   struct Student* next;
};


struct Student* insert(struct Student* head, char name[], int marks) {
   struct Student* newNode = (struct Student*)malloc(sizeof(struct Student));
   strcpy(newNode->name, name);
   newNode->marks = marks;
   newNode->next = NULL;

   if (head == NULL) {
       return newNode;
   }

   struct Student* temp = head;
   while (temp->next != NULL) {
       temp = temp->next;
   }
   temp->next = newNode;

   return head;
}

// Print list
void printList(struct Student* head) {
   struct Student* temp = head;
   while (temp != NULL) {
       printf("%s (%d) -> ", temp->name, temp->marks);
       temp = temp->next;
   }
   printf("NULL\n");
}

// Update marks
void updateMarks(struct Student* head, char name[], int newMarks) {
   struct Student* temp = head;
   while (temp != NULL) {
       if (strcmp(temp->name, name) == 0) {
           temp->marks = newMarks;
           printf("Marks successfully updated.\n");
           return;
       }
       temp = temp->next;
   }
   printf("Student not found.\n");
}


struct Student* deleteFirst(struct Student* head) {
   if (head == NULL) return NULL;

   struct Student* temp = head;
   head = head->next;
   free(temp);

   return head;
}


int main() {
   struct Student* head = NULL;

//Insert 3 students
   head = insert(head, "Ayan", 12);
   head = insert(head, "Nabila", 15);
   head = insert(head, "Farhan", 18);

   printf("List after insertion:\n");
   printList(head);

//Print list
   printf("\nAll students:\n");
   printList(head);

//Update marks
   updateMarks(head, "Nabila", 20);

   printf("\nAfter updating Nabila's marks:\n");
   printList(head);

//Delete student record
   head = deleteFirst(head);

   printf("\nAfter deleting first student:\n");
   printList(head);

   return 0;
}
